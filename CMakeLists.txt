cmake_minimum_required(VERSION 3.22)
project(GlucoidStudio VERSION 1.5.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network DBus Test Graphs Concurrent)
find_package(KF6Wallet REQUIRED)
find_package(SQLite3 REQUIRED)
qt_standard_project_setup(REQUIRES 6.10)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=.")
endif()
add_library(studio-core STATIC src/glucose.cpp src/glucoseanalysis.cpp src/dailyanalysis.cpp src/transportcore.cpp src/domain.cpp src/store.cpp src/transport.cpp src/analysis.cpp src/importer.cpp src/response.cpp src/observation.cpp src/automatic.cpp src/targetmodel.cpp)
target_sources(studio-core PRIVATE src/daychart.cpp src/wireguard.cpp)
target_sources(studio-core PRIVATE src/nightscout.cpp src/nightscoutstore.cpp)
target_include_directories(studio-core PUBLIC src)
target_link_libraries(studio-core PUBLIC Qt6::Core Qt6::Network Qt6::DBus SQLite::SQLite3)
target_compile_options(studio-core PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(glucoid-studio src/main.cpp src/backend.cpp src/backend.h)
qt_add_resources(glucoid-studio studio_icon PREFIX "/" FILES packaging/fi.reijo.glucoidstudio.png)
target_compile_options(glucoid-studio PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(glucoid-studio PRIVATE studio-core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Graphs Qt6::Concurrent KF6::Wallet)
set(STUDIO_LOCAL_DATA_ROOT "" CACHE PATH "Optional local development data root; empty uses XDG application data")
target_compile_definitions(glucoid-studio PRIVATE GLUCOID_VERSION="${PROJECT_VERSION}")
if(STUDIO_LOCAL_DATA_ROOT)
  target_compile_definitions(glucoid-studio PRIVATE STUDIO_LOCAL_DATA_ROOT="${STUDIO_LOCAL_DATA_ROOT}")
endif()
include(GNUInstallDirs)
install(TARGETS glucoid-studio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/fi.reijo.glucoidstudio.desktop DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
install(FILES packaging/fi.reijo.glucoidstudio.svg DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/scalable/apps)
install(FILES packaging/fi.reijo.glucoidstudio.png DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/256x256/apps)
install(FILES LICENSE packaging/THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/glucoid-studio)
install(DIRECTORY packaging/licenses/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/glucoid-studio/licenses)
qt_add_qml_module(glucoid-studio URI GlucoidStudio VERSION 1.0 QML_FILES qml/chartsegments.js qml/ConnectionEditor.qml qml/theme.js qml/Main.qml qml/DailyAnalysis.qml qml/ResearchMain.qml qml/ResponsePanel.qml qml/LegacyMain.qml qml/GlucoseChart.qml qml/CurrentDayChart.qml qml/ProfileChart.qml qml/ProfileEditor.qml qml/Disclosure.qml qml/MetricCard.qml qml/Legend.qml qml/ReviewCheckBox.qml)
include(CTest)

  target_link_libraries(response-simulations PRIVATE studio-core)
