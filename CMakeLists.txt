cmake_minimum_required(VERSION 3.22)
project(GlucoidStudio VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network DBus Graphs Concurrent LinguistTools)
find_package(KF6Wallet REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(SQLite3 REQUIRED)
qt_standard_project_setup(REQUIRES 6.10)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=.")
endif()
add_library(studio-core STATIC
  src/glucose.cpp src/glucoseanalysis.cpp src/dailyanalysis.cpp src/daychart.cpp
  src/domain.cpp src/store.cpp src/observation.cpp src/importer.cpp
  src/transportcore.cpp src/transport.cpp src/wireguard.cpp
  src/nightscout.cpp src/nightscoutstore.cpp src/databasebackup.cpp)
target_include_directories(studio-core PUBLIC src)
target_link_libraries(studio-core PUBLIC Qt6::Core Qt6::Network Qt6::DBus Qt6::Concurrent SQLite::SQLite3)
target_compile_options(studio-core PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(glucoid-studio src/main.cpp src/backend.cpp src/backend.h src/language.cpp src/language.h)
qt_add_resources(glucoid-studio studio_icon PREFIX "/" FILES packaging/fi.reijo.glucoidstudio.png)
target_compile_options(glucoid-studio PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(glucoid-studio PRIVATE studio-core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Graphs Qt6::Concurrent KF6::Wallet KF6::I18n)
set(STUDIO_LOCAL_DATA_ROOT "" CACHE PATH "Optional local development data root; empty uses XDG application data")
target_compile_definitions(glucoid-studio PRIVATE GLUCOID_VERSION="${PROJECT_VERSION}")
if(STUDIO_LOCAL_DATA_ROOT)
  target_compile_definitions(glucoid-studio PRIVATE STUDIO_LOCAL_DATA_ROOT="${STUDIO_LOCAL_DATA_ROOT}")
endif()
include(GNUInstallDirs)
install(TARGETS glucoid-studio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/fi.reijo.glucoidstudio.desktop DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
install(FILES packaging/fi.reijo.glucoidstudio.svg DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/scalable/apps)
install(FILES packaging/fi.reijo.glucoidstudio.png DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/256x256/apps)
install(FILES LICENSE packaging/THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/glucoid-studio)
install(DIRECTORY packaging/licenses/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/doc/glucoid-studio/licenses)
qt_add_qml_module(glucoid-studio URI GlucoidStudio VERSION 1.0 QML_FILES
  qml/chartsegments.js qml/theme.js qml/Main.qml qml/DailyAnalysis.qml
  qml/ConnectionEditor.qml qml/LanguageSelector.qml qml/GlucoseChart.qml
  qml/CurrentDayChart.qml qml/ProfileChart.qml qml/Disclosure.qml
  qml/MetricCard.qml qml/Legend.qml)
set(studio_translation_targets glucoid-studio)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(studio-test-support STATIC tests/storefixtures.cpp)
  target_link_libraries(studio-test-support PUBLIC studio-core)
  function(studio_test target name source)
    qt_add_executable(${target} ${source} ${ARGN})
    target_link_libraries(${target} PRIVATE studio-test-support Qt6::Test)
    target_include_directories(${target} PRIVATE tests)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(${target} PRIVATE STUDIO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" STUDIO_APP_VERSION="${PROJECT_VERSION}")
    add_test(NAME ${name} COMMAND ${target})
    set_tests_properties(${name} PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endfunction()
  function(studio_backend_test target name source)
    studio_test(${target} ${name} ${source} src/backend.cpp src/backend.h ${ARGN})
    target_link_libraries(${target} PRIVATE Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Graphs KF6::Wallet KF6::I18n)
  endfunction()
  studio_test(security-audit-tests studio-security-audit tests/security/test_security.cpp)
  studio_test(dailyanalysis-tests studio-dailyanalysis tests/test_dailyanalysis.cpp)
  studio_test(studio-tests studio-core tests/test_studio.cpp)
  studio_test(database-backup-tests studio-database-backup tests/test_databasebackup.cpp)
  target_compile_definitions(studio-tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  studio_backend_test(nightscout-tests studio-nightscout tests/test_nightscout.cpp)
  find_program(OPENSSL_TEST_EXECUTABLE openssl REQUIRED)
  target_compile_definitions(nightscout-tests PRIVATE OPENSSL_TEST_EXECUTABLE="${OPENSSL_TEST_EXECUTABLE}")
  studio_backend_test(debug-fixes-tests studio-debug-fixes tests/test_debug_fixes.cpp)
  studio_backend_test(backend-tests studio-backend tests/test_backend.cpp)
  studio_backend_test(product-scope-tests studio-product-scope tests/test_product_scope.cpp)
  studio_backend_test(main-ui-tests studio-main-ui tests/test_main_ui.cpp src/language.cpp src/language.h)
  studio_backend_test(language-tests studio-language tests/test_language.cpp src/language.cpp src/language.h)
  list(APPEND studio_translation_targets main-ui-tests language-tests)
  qt_add_executable(nightscout-live-tests tests/test_nightscout_live.cpp)
  target_link_libraries(nightscout-live-tests PRIVATE studio-core)
  add_test(NAME studio-nightscout-live COMMAND nightscout-live-tests)
  set_tests_properties(studio-nightscout-live PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME studio-product-binary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_product_binary.py $<TARGET_FILE:glucoid-studio>)
  set_tests_properties(studio-product-binary PROPERTIES TIMEOUT 25)
endif()
qt_add_translations(TARGETS ${studio_translation_targets}
  TS_FILES i18n/studio_en.ts i18n/studio_fi.ts
  RESOURCE_PREFIX /i18n
  MERGE_QT_TRANSLATIONS QT_TRANSLATION_CATALOGS qtbase qtdeclarative)
